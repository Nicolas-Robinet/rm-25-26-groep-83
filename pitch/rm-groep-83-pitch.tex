\section{Internetcensuur in België.}
%==============================================================================

\begin{frame}
    \frametitle{Waarom dit onderwerp telt.}

    \begin{columns}[t]
        \column{.33\textwidth}
        \textbf{Maatschappelijk}

        \medskip
        Vrije toegang tot informatie raakt aan grondrechten en de
        werking van de democratie.

        \column{.33\textwidth}
        \textbf{Binnen de specialisatie}

        \medskip
        Netwerkblokkades, DNS-filtering en omzeiling vormen de kern van
        System \& Network Engineering.

        \column{.33\textwidth}
        \textbf{Actueel}

        \medskip
        Censuur op het web groeit wereldwijd; ook in België worden
        websites geblokkeerd zonder veel ruchtbaarheid.
    \end{columns}
\end{frame}

\subsection{Probleemstelling.}

\begin{frame}
    \frametitle{Probleemstelling.}

    Censuur op het internet is wereldwijd een groeiend thema.
    \textbf{Ook in België worden websites geblokkeerd}, maar wie beslist
    wat, en op welke basis?

    \bigskip

    Het kernprobleem is de \textbf{onduidelijkheid en beperkte
    transparantie} rond de toepassing van internetcensuur in België.

    \bigskip
    \bigskip

    \textcolor{hggrey}{\textbf{Onderbouwd door bronnen.}} TechPulse
    (2015) toont aan dat de overheid meer websites blokkeert dan ze
    toegeeft.
\end{frame}

\subsection{Onderzoeksvragen.}

\begin{frame}
    \frametitle{Hoofdonderzoeksvraag.}

    \begin{quote}
        {\large In welke mate beïnvloedt de Belgische overheid het online
        gedrag van burgers door het al dan niet toepassen van
        internetcensuur?}
    \end{quote}

    \bigskip

\end{frame}

\begin{frame}
    \frametitle{Deelvragen.}

    \begin{columns}[t]
        \column{.5\textwidth}
        \textbf{Probleemdomein}
        \begin{itemize}
            \item Welke vormen van internetcensuur worden momenteel in
                  België toegepast?
            \item Welke instanties beoordelen welke online content
                  gecensureerd moet worden?
            \item Welk online gedrag wordt aantoonbaar beïnvloed door
                  overheidsinterventie?
        \end{itemize}

        \column{.5\textwidth}
        \textbf{Oplossingsdomein}
        \begin{itemize}
            \item Welke technische middelen (VPN, Tor) gebruiken burgers
                  om censuur te omzeilen en hoe effectief zijn die?
        \end{itemize}
    \end{columns}
\end{frame}

\section{Online veiligheid van kinderen in digitale games.}
%==============================================================================

\begin{frame}
    \frametitle{Waarom dit onderwerp telt.}

    \begin{columns}[t]
        \column{.33\textwidth}
        \textbf{Maatschappelijk}

        \medskip
        Kinderen gamen steeds vroeger; hun veiligheid online is een brede
        zorg voor ouders, scholen en beleid.

        \column{.33\textwidth}
        \textbf{Interprofessioneel}

        \medskip
        Het thema verbindt technologie, onderwijs, recht en welzijn rond
        één kwetsbare doelgroep.

        \column{.33\textwidth}
        \textbf{Actueel}

        \medskip
        Loot boxes, grooming en cyberpesten halen geregeld het nieuws;
        regelgeving holt de praktijk achterna.
    \end{columns}
\end{frame}

\subsection{Probleemstelling.}

\begin{frame}
    \frametitle{Probleemstelling.}

    Kinderen en jongeren gamen steeds vroeger en intensiever.
    \textbf{Toch blijven ze blootgesteld aan online risico's}: grooming,
    cyberpesten, gewelddadige inhoud en loot boxes.

    \bigskip

    Ondanks classificatiesystemen en ouderlijke controles is er een
    \textbf{beperkt en onduidelijk toegepast aanbod aan
    veiligheidsmaatregelen} voor minderjarigen.

    \bigskip
    \bigskip

    \textcolor{hggrey}{\textbf{Onderbouwd door bronnen.}} Bloomberg
    (Carville \& Chen, 2024) en UNICEF (2024) documenteren de risico's
    voor kinderen online.
\end{frame}

\subsection{Onderzoeksvragen.}

\begin{frame}
    \frametitle{Hoofdonderzoeksvraag.}

    \begin{quote}
        {\large In welke mate worden kinderen en jongeren in België
        beschermd tegen online risico's binnen digitale games?}
    \end{quote}

    \bigskip

    
\end{frame}

\begin{frame}
    \frametitle{Deelvragen.}

    \begin{columns}[t]
        \column{.5\textwidth}
        \textbf{Probleemdomein}
        \begin{itemize}
            \item Welke online risico's (loot boxes, grooming,
                  cyberpesten, gewelddadige inhoud) komen voor in games
                  die minderjarigen spelen?
            \item Hoe effectief zijn bestaande classificatiesystemen
                  zoals PEGI in het beschermen van minderjarigen?
        \end{itemize}

        \column{.5\textwidth}
        \textbf{Oplossingsdomein}
        \begin{itemize}
            \item Welke technische veiligheidsmaatregelen (ouderlijk
                  toezicht, chatfilters, meldknoppen) bieden
                  gameontwikkelaars aan en hoe toegankelijk zijn die?
        \end{itemize}
    \end{columns}
\end{frame}
